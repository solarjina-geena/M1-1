# ============================================================
# [MOTIE 월별 자동차 시장 데이터]
# 산업통상부 자동차산업 월별 MASTER를 시장 MART에 연결합니다.
#
# 데이터:
#   data/raw/motie/motie_monthly_master.csv
#
# 컬럼:
#   period      : YYYY-MM
#   production  : 생산량
#   domestic    : 내수 판매량
#   export      : 수출량
#
# 중요:
#   일부 월은 원자료가 없으므로 임의 보간하지 않습니다.
#   실제 확보된 월만 분석에 사용합니다.
# ============================================================

MOTIE_MONTHLY_MASTER = (
    Path(__file__).resolve().parents[1]
    / "data"
    / "raw"
    / "motie"
    / "motie_monthly_master.csv"
)

def build_market_mart() -> pd.DataFrame:
    production = load_latest_vehicle_production_processed()

    rows = []
    # ========================================================
    # [1] MOTIE 월별 자동차 시장 데이터 연결
    # ========================================================
    # 우리가 검증/정리한 월별 MASTER를 읽습니다.
    #
    # period     : 기준월 (예: 2026-08)
    # production : 국내 자동차 생산량
    # domestic   : 국내 자동차 내수 판매량
    # export     : 국내 자동차 수출량
    #
    # 결측월은 임의로 보간하지 않습니다.
    # 실제 데이터가 존재하는 월만 시장 MART에 저장합니다.
    # ========================================================

    if MOTIE_MONTHLY_MASTER.exists():
        motie = pd.read_csv(
            MOTIE_MONTHLY_MASTER,
            dtype={"period": str},
        )

        # ----------------------------------------------------
        # [1-1] 숫자 컬럼 정리
        # 문자 등이 섞여 있어도 숫자로 안전하게 변환합니다.
        # ----------------------------------------------------
        for col in ["production", "domestic", "export"]:
            motie[col] = pd.to_numeric(
                motie[col],
                errors="coerce",
            )

        # ----------------------------------------------------
        # [1-2] MASTER 컬럼을 대시보드용 지표명으로 변환
        # ----------------------------------------------------
        motie_metrics = {
            "production": "VEHICLE_PRODUCTION_TOTAL",
            "domestic": "VEHICLE_DOMESTIC_TOTAL",
            "export": "VEHICLE_EXPORT_TOTAL",
        }

        # ----------------------------------------------------
        # [1-3] 월별 데이터를 시장 MART 형식으로 저장
        # ----------------------------------------------------
        for source_col, metric_name in motie_metrics.items():

            for _, motie_row in motie.iterrows():
                value = motie_row.get(source_col)

                # 값이 없는 월은 만들어내지 않고 건너뜁니다.
                if pd.isna(value):
                    continue

                rows.append(
                    {
                        "period": motie_row["period"],
                        "market_scope": "KOREA",
                        "metric": metric_name,
                        "value": float(value),
                        "unit": "EA",
                        "source_id": motie_row.get(
                            "source_file",
                            "motie_monthly_master.csv",
                        ),
                        "source_key": "motie",
                        "estimate_flag": "ACTUAL",
                        "confidence_grade": "",
                        "model_version": "",
                    }
                )

    # KOSIS 자동차 생산량을 시장 기준 데이터로 변환
    for _, row in production.iterrows():
        year = row.get("year")
        qty = row.get("production_qty")

        if pd.isna(year):
            continue

        country_code = str(row.get("country_code", ""))
        category = str(row.get("vehicle_category", ""))

        # ====================================================
        # [2] KOSIS 기존 데이터의 사용 범위
        # ====================================================
        # GLOBAL:
        #   기존 KOSIS 연간 자동차 생산 데이터를 계속 사용합니다.
        #
        # KOREA:
        #   위에서 MOTIE 월별 MASTER를 새로 연결했으므로
        #   기존 KOSIS 한국 연간 데이터는 넣지 않습니다.
        #
        # 이유:
        #   KOSIS 한국 연간값 + MOTIE 한국 월별값이
        #   동시에 들어가는 중복을 방지하기 위해서입니다.
        # ====================================================

        if country_code == "0":
            scope = "GLOBAL"

        elif country_code == "1005":
            continue

        else:
            continue
        
        rows.append(
            {
                "period": f"{int(year)}-12",
                "market_scope": scope,
                "metric": f"VEHICLE_PRODUCTION_{category}",
                "value": (
                    float(qty)
                    if not pd.isna(qty)
                    else pd.NA
                ),
                "unit": "EA",
                "source_id": row.get("source_id"),
                "source_key": "kosis",
                "estimate_flag": row.get(
                    "estimate_flag",
                    "ACTUAL",
                ),
                "confidence_grade": "",
                "model_version": "",
            }
        )

    # 제품/당사 공급 기준은 Master를 단일 원본으로 사용
    assumptions = load_market_assumption_master()

    metric_map = {
        "coils_per_system": "COILS_PER_SYSTEM",
        "company_iv_per_system": "COMPANY_IV_PER_SYSTEM",
        "company_ov_per_system": "COMPANY_OV_PER_SYSTEM",
        "company_hsv_per_system": "COMPANY_HSV_PER_SYSTEM",
    }

    model_rows = []

    for _, row in assumptions.iterrows():
        model_id = row.get("model_id")
        parameter_name = row.get("parameter_name")

        if pd.isna(model_id) or parameter_name not in metric_map:
            continue

        value = pd.to_numeric(
            row.get("parameter_value"),
            errors="coerce",
        )

        model_rows.append(
            {
                "period": "2026-09",
                "market_scope": str(model_id),
                "metric": metric_map[parameter_name],
                "value": value if not pd.isna(value) else pd.NA,
                "unit": row.get("unit"),
                "source_id": row.get("source_id"),
                "source_key": "market_assumption_master",
                "estimate_flag": row.get("estimate_flag"),
                "confidence_grade": row.get("confidence_grade"),
                "model_version": row.get("model_version"),
            }
        )

    rows.extend(model_rows)

    mart = pd.DataFrame(rows)

    for column in MARKET_MONTHLY_COLUMNS:
        if column not in mart.columns:
            mart[column] = pd.NA

    mart = mart[MARKET_MONTHLY_COLUMNS]

    return mart


def save_market_mart() -> dict:
    mart = build_market_mart()

    OUTPUT_PATH.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    mart.to_parquet(
        OUTPUT_PATH,
        index=False,
    )

    return {
        "output_file": str(OUTPUT_PATH),
        "row_count": len(mart),
        "actual_count": int(
            (mart["estimate_flag"] == "ACTUAL").sum()
        ),
        "estimated_count": int(
            (mart["estimate_flag"] == "ESTIMATED").sum()
        ),
    }


if __name__ == "__main__":
    result = save_market_mart()

    print("Market mart build completed")
    print("OUTPUT_FILE:", result["output_file"])
    print("ROW_COUNT:", result["row_count"])
    print("ACTUAL:", result["actual_count"])
    print("ESTIMATED:", result["estimated_count"])

def build_technology_signal_mart() -> pd.DataFrame:
    if not SOURCE_PATH.exists():
        return pd.DataFrame(columns=COLUMNS)

    df = pd.read_csv(
        SOURCE_PATH,
        dtype="string",
    )

    for column in COLUMNS:
        if column not in df.columns:
            df[column] = pd.NA

    df = df[COLUMNS].copy()

    df["forecast_value"] = pd.to_numeric(
        df["forecast_value"],
        errors="coerce",
    )

    df["forecast_year"] = pd.to_numeric(
        df["forecast_year"],
        errors="coerce",
    ).astype("Int64")

    # signal_date가 YYYY 형식이어도 정렬 가능하도록 보조 연도 생성
    df["signal_year"] = pd.to_numeric(
        df["signal_date"].str[:4],
        errors="coerce",
    ).astype("Int64")

    df = df.sort_values(
        ["signal_year", "company", "signal_type"],
        ascending=[False, True, True],
        kind="stable",
    ).reset_index(drop=True)

    return df

import hashlib

import pandas as pd

RAW_DIR = Path(PATH_CONFIG["raw"]) / "technical"
SOURCE_LOG_PATH = Path(PATH_CONFIG["metadata"]) / "source_log.csv"


def _safe_filename(text: str) -> str:
    allowed = (
        "-_.() "
        "abcdefghijklmnopqrstuvwxyz"
        "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
        "0123456789"
    )
    cleaned = "".join(
        c if c in allowed else "_" for c in text
    )
    cleaned = "_".join(cleaned.split())
    return cleaned.strip("_") or "technical_source"


def _guess_extension(
    url: str,
    content_type: str | None,
) -> str:
    suffix = Path(urlparse(url).path).suffix.lower()

    if suffix and len(suffix) <= 6:
        return suffix

    if content_type:
        mime = content_type.split(";")[0].strip()
        guessed = mimetypes.guess_extension(mime)
        if guessed:
            return guessed

    return ".html"


def _sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def collect_technical_source(
    url: str,
    title: str,
    notes: str = "",
) -> dict:
    RAW_DIR.mkdir(parents=True, exist_ok=True)

    response = requests.get(
        url,
        timeout=60,
        headers={
            "User-Agent": (
                "Mozilla/5.0 ABS-Market-Intelligence "
                "technical-source-collector"
            )
        },
    )
    response.raise_for_status()

    collected_at = datetime.now()
    timestamp = collected_at.strftime("%Y%m%d%H%M%S")

    extension = _guess_extension(
        url,
        response.headers.get("Content-Type"),
    )

    filename = (
        f"{timestamp}_{_safe_filename(title)}"
        f"{extension}"
    )

    raw_path = RAW_DIR / filename
    raw_path.write_bytes(response.content)

    digest = _sha256(response.content)
    source_id = f"TECH_{timestamp}"

    source_info = SOURCES.get(
        "technical",
        {
            "name": "공개 기술자료 / 제조사 공식자료",
        },
    )

    log_row = {
        "source_id": source_id,
        "source_key": "technical",
        "source_name": source_info["name"],
        "basis_date": "",
        "collected_at": collected_at.isoformat(
            timespec="seconds"
        ),
        "source_url": url,
        "raw_file_path": str(raw_path),
        "sha256": digest,
        "status": "SUCCESS",
        "notes": notes or title,
    }

    if SOURCE_LOG_PATH.exists():
        log_df = pd.read_csv(
            SOURCE_LOG_PATH,
            dtype="string",
        )
    else:
        log_df = pd.DataFrame(
            columns=list(log_row.keys())
        )

    log_df = pd.concat(
        [
            log_df,
            pd.DataFrame([log_row]),
        ],
        ignore_index=True,
    )

    log_df.to_csv(
        SOURCE_LOG_PATH,
        index=False,
        encoding="utf-8-sig",
    )

    return {
        "source_id": source_id,
        "raw_file": str(raw_path),
        "sha256": digest,
        "bytes": len(response.content),
        "content_type": response.headers.get(
            "Content-Type"
        ),
    }


if __name__ == "__main__":
    print("technical_source_collector 준비 완료")

METRIC_CANDIDATES = {
    "생산": [
        "VEHICLE_PRODUCTION_TOTAL",
    ],
    "내수": [
        "VEHICLE_DOMESTIC_TOTAL",
        "VEHICLE_DOMESTIC_SALES_TOTAL",
        "DOMESTIC_SALES_TOTAL",
        "VEHICLE_SALES_DOMESTIC_TOTAL",
    ],
    "수출": [
        "VEHICLE_EXPORT_TOTAL",
        "EXPORT_TOTAL",

    ],
}


def find_metric(label, scope="KOREA"):
    """현재 MART에 실제로 존재하는 metric 이름을 자동으로 찾습니다."""
    if market.empty or "metric" not in market.columns:
        return None

    view = market.copy()

    if "market_scope" in view.columns:
        view = view[
            view["market_scope"].astype(str).str.upper()
            == str(scope).upper()
        ]

    available = set(view["metric"].astype(str))

    for candidate in METRIC_CANDIDATES[label]:
        if candidate in available:
            return candidate

    keyword = {
        "생산": "PRODUCTION",
        "내수": "DOMESTIC",
        "수출": "EXPORT",
    }[label]

    matches = [
        x for x in available
        if keyword in x.upper() and "TOTAL" in x.upper()
    ]

    return sorted(matches)[0] if matches else None


def monthly_series(label, scope="KOREA"):
    """생산/내수/수출 월별 실제값을 반환합니다."""
    metric = find_metric(label, scope)

    if metric is None or market.empty:
        return pd.DataFrame()

    df = market[
        (market["market_scope"].astype(str).str.upper() == scope.upper())
        & (market["metric"].astype(str) == metric)
    ].copy()

    if df.empty:
        return df

    df["date"] = pd.to_datetime(
        df["period"].astype(str),
        format="%Y-%m",
        errors="coerce",
    )
    df["value"] = pd.to_numeric(df["value"], errors="coerce")

    df = (
        df.dropna(subset=["date", "value"])
        .sort_values("date")
        .drop_duplicates(subset=["date"], keep="last")
        .reset_index(drop=True)
    )

    df["series"] = label
    return df[["date", "value", "series"]]


def calendar_features(df):
    """
    달력 기준 시계열 분석.
    없는 월은 임의로 채우지 않습니다.
    """
    if df.empty:
        return pd.DataFrame()

    x = (
        df[["date", "value"]]
        .drop_duplicates("date", keep="last")
        .set_index("date")
        .sort_index()
    )

    full = pd.date_range(x.index.min(), x.index.max(), freq="MS")
    x = x.reindex(full)
    x.index.name = "date"

    x["mom_pct"] = x["value"].pct_change(1, fill_method=None) * 100
    x["yoy_pct"] = x["value"].pct_change(12, fill_method=None) * 100
    x["ma3"] = x["value"].rolling(3, min_periods=3).mean()
    x["std12"] = x["value"].rolling(12, min_periods=12).std()
    x["missing"] = x["value"].isna()

    return x.reset_index()


def latest_actual(df):
    if df.empty:
        return None
    x = df.dropna(subset=["value"]).sort_values("date")
    return None if x.empty else x.iloc[-1]


def period_start_date(mode, latest_date):
    if mode == "최근 3년":
        return latest_date - pd.DateOffset(years=3) + pd.DateOffset(months=1)
    if mode == "최근 5년":
        return latest_date - pd.DateOffset(years=5) + pd.DateOffset(months=1)
    if mode == "최근 10년":
        return latest_date - pd.DateOffset(years=10) + pd.DateOffset(months=1)
    return pd.Timestamp("1900-01-01")


def fmt_count(v):
    if v is None or pd.isna(v):
        return "-"
    return f"{v:,.0f}대"


def fmt_pct(v):
    if v is None or pd.isna(v):
        return "-"
    return f"{v:+.1f}%"


def safe_yoy(df):
    """최신월과 정확히 12개월 전을 비교합니다."""
    f = calendar_features(df)
    if f.empty:
        return None
    actual = f.dropna(subset=["value"])
    if actual.empty:
        return None
    return actual.iloc[-1]["yoy_pct"]


def annual_total(df, year):
    if df.empty:
        return None
    x = df[df["date"].dt.year == year]["value"]
    if x.empty:
        return None
    return float(x.sum())


def annual_change(df, year):
    now = annual_total(df, year)
    prev = annual_total(df, year - 1)
    if now is None or prev in (None, 0):
        return None
    return (now / prev - 1) * 100

def current_market_chart(compare_mode, selected_year):
    rows = []

    for label in ["생산", "내수", "수출"]:
        df = monthly_series(label)
        if df.empty:
            continue

        if compare_mode == "전년 대비":
            # 선택 연도의 월별 합계와 전년 같은 기간을 비교
            current = df[df["date"].dt.year == selected_year].copy()
            previous = df[df["date"].dt.year == selected_year - 1].copy()

            if current.empty:
                continue

            max_month = int(current["date"].dt.month.max())
            previous = previous[previous["date"].dt.month <= max_month]

            current_value = float(current["value"].sum())
            previous_value = (
                float(previous["value"].sum())
                if not previous.empty
                else None
            )

            change = (
                (current_value / previous_value - 1) * 100
                if previous_value not in (None, 0)
                else None
            )

            rows.append({
                "구분": label,
                "값": current_value,
                "증감률": change,
                "비교": f"{selected_year}년 1~{max_month}월",
            })

        else:
            value = annual_total(df, selected_year)
            if value is None:
                continue
            rows.append({
                "구분": label,
                "값": value,
                "증감률": annual_change(df, selected_year),
                "비교": f"{selected_year}년",
            })

    chart_df = pd.DataFrame(rows)

    if chart_df.empty:
        st.info("선택한 기간의 시장 데이터가 없습니다.")
        return

    bars = (
        alt.Chart(chart_df)
        .mark_bar(cornerRadiusTopLeft=5, cornerRadiusTopRight=5)
        .encode(
            x=alt.X("구분:N", title=None, sort=["생산", "내수", "수출"]),
            y=alt.Y("값:Q", title="대수", axis=alt.Axis(format="~s")),
            color=alt.Color(
                "구분:N",
                legend=None,
                scale=alt.Scale(
                    domain=["생산", "내수", "수출"],
                    range=["#355C9A", "#2F8F83", "#D48645"],
                ),
            ),
            tooltip=[
                alt.Tooltip("구분:N", title="구분"),
                alt.Tooltip("비교:N", title="기간"),
                alt.Tooltip("값:Q", title="대수", format=",.0f"),
                alt.Tooltip("증감률:Q", title="증감률", format="+.1f"),
            ],
        )
        .properties(height=245)
    )

    st.altair_chart(bars, use_container_width=True)

        return (
            "계산 가능한 데이터가 없습니다.",
            "현재 자료만으로 해석하기 어렵습니다.",
            "시장 데이터 연결 상태를 먼저 확인합니다.",
        )

    f = calendar_features(df)
    actual = f.dropna(subset=["value"]).sort_values("date")

    latest = actual.iloc[-1]
    yoy_value = latest["yoy_pct"]

    if question == "수출 변동성이 커진 시점은 언제인가?":
        vol = f.dropna(subset=["std12"])
        if not vol.empty:
            peak = vol.loc[vol["std12"].idxmax()]
            fact = (
                f"{peak['date']:%Y-%m}에 12개월 변동성이 "
                f"{peak['std12']:,.0f}대로 가장 크게 나타났습니다."
            )
        else:
            fact = "연속 12개월 자료가 부족해 변동성 최대시점을 계산할 수 없습니다."

        why = (
            "변동성 확대는 시장 방향 자체보다 월별 충격이 커졌다는 신호입니다. "
            "당시 수출지역·생산차질·외부 사건과 함께 확인해야 합니다."
        )
        action = (
            "수출 변동이 큰 구간과 기술전환 신호를 겹쳐 보고 "
            "통합제동·신기술 적용시장 확대 여부를 추가 확인합니다."
        )
        return fact, why, action

    fact = (
        f"최근 {subject}은 {latest['date']:%Y-%m} "
        f"{latest['value']:,.0f}대이며, "
        f"전년 동월 대비 {fmt_pct(yoy_value)}입니다."
    )

    if pd.isna(yoy_value):
        why = "비교 기준월 데이터가 없어 전년 동월 변화만으로 방향을 단정하지 않습니다."
    elif yoy_value >= 0:
        why = "최근 값은 전년 동월보다 높습니다. 회복이 일시적인지 추세적인지는 이동평균과 함께 봐야 합니다."
    else:
        why = "최근 값은 전년 동월보다 낮습니다. 계절성·조업일수·수요 변화 등 외부 요인 확인이 필요합니다."

    action = (
        "전체 자동차 수량만 보지 않고 ABS → ESC → 통합제동 → "
        "Brake-by-Wire 전환과 연결해 당사 접근 가능 시장을 확인합니다."
    )

    return fact, why, action

def render_technology_evidence():
    if signals.empty:
        st.info("등록된 기술 신호가 없습니다.")
        return

    view = signals.copy()

    sort_cols = [
        c for c in ["signal_year", "company"]
        if c in view.columns
    ]
    if sort_cols:
        view = view.sort_values(
            sort_cols,
            ascending=[False] * len(sort_cols),
        )

    view = view.head(4)

    cols = st.columns(2)

    for i, (_, row) in enumerate(view.iterrows()):
        with cols[i % 2]:
            company = str(row.get("company", "기술자료"))
            year = row.get("signal_year", "")
            event = str(row.get("event", "")).strip()
            status = str(row.get("verification_status", "")).strip()
            source_url = row.get("source_url", None)

            st.markdown(
                f"""
                <div class="source-card">
                    <strong>{company}</strong>
                    <span style="color:#8a93a3;"> · {year}</span><br>
                    {event}
                </div>
                """,
                unsafe_allow_html=True,
            )

            if status == "VERIFIED":
                st.caption("공식자료 · 검증 완료")

            if source_url is not None and str(source_url).startswith("http"):
                st.link_button(
                    "근거자료 보기 ↗",
                    str(source_url),
                    use_container_width=True,
                )
