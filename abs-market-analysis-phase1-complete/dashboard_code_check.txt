import pandas as pd
import altair as alt
import streamlit as st

from processors.time_series_analysis import (
    add_timeseries_features,
    summarize_timeseries,
    compare_recent_vs_longterm,
    detect_frequency,

def _pct(value):
    if value is None or pd.isna(value):
        return "확인 불가"
    return f"{value:+.2f}%"


def _million(value):
    if value is None or pd.isna(value):
        return "확인 불가"
    return f"{value / 1_000_000:.2f}M대"


def _prepare(df):
    if df.empty:
        return df

    result = df.copy()

    if "date" not in result.columns:
        result["date"] = pd.to_datetime(
            result["period"],
            format="%Y-%m",
            errors="coerce",
        )

    if "period_label" not in result.columns:
        result["period_label"] = (
            result["date"].dt.strftime("%Y-%m")
        )

    result["year"] = result["date"].dt.year
    result["month"] = result["date"].dt.month

    return (
        result.dropna(subset=["date", "value"])
        .sort_values("date")
        .reset_index(drop=True)
    )


def _trend_chart(data, title):
    frequency = detect_frequency(data)
    analyzed = add_timeseries_features(data)

    if frequency == "MONTHLY":
        ma_col = "ma_12m"
        ma_name = "12개월 이동평균"
    else:
        ma_col = "ma_3y"
        ma_name = "3년 이동평균"

    chart_data = analyzed[
        ["date", "value", ma_col]
    ].copy()

    chart_data = chart_data.rename(
        columns={
            "value": "실제값",
            ma_col: ma_name,
        }
    )

    long = chart_data.melt(
        id_vars=["date"],
        var_name="구분",
        value_name="value",
    )

    base = (
        alt.Chart(long)
        .mark_line(
            strokeWidth=2.5,
            point=True,
        )
        .encode(
            x=alt.X(
                "date:T",
                title=None,
                axis=alt.Axis(format="%Y-%m"),
            ),
            y=alt.Y(
                "value:Q",
                title="자동차 생산대수",
                axis=alt.Axis(format="~s"),
                scale=alt.Scale(zero=False),
            ),
            color=alt.Color(
                "구분:N",
                title=None,
            ),
            tooltip=[
                alt.Tooltip(
                    "date:T",
                    title="기간",
                    format="%Y-%m",
                ),
                alt.Tooltip(
                    "구분:N",
                    title="구분",
                ),
                alt.Tooltip(
                    "value:Q",
                    title="생산대수",
                    format=",.0f",
                ),
            ],
        )
        .properties(
            height=300,
            title=title,
        )
        .interactive()
    )

    return base


def _change_chart(data):
    analyzed = add_timeseries_features(data)
    frequency = detect_frequency(analyzed)

    change_col = (
        "yoy_pct"
        if frequency == "ANNUAL"
        else "mom_pct"
    )

    label = (
        "연간 변화율 (YoY)"
        if frequency == "ANNUAL"
        else "전월 대비 변화율 (MoM)"
    )

    chart_data = analyzed[
        ["date", change_col]
    ].dropna().copy()

    chart_data["방향"] = chart_data[
        change_col
    ].apply(
        lambda x: "증가" if x >= 0 else "감소"
    )

    zero = (
        alt.Chart(pd.DataFrame({"y": [0]}))
        .mark_rule(
            strokeDash=[4, 4],
        )
        .encode(y="y:Q")
    )

    bars = (
        alt.Chart(chart_data)
        .mark_bar(
            cornerRadiusTopLeft=3,
            cornerRadiusTopRight=3,
        )
        .encode(
            x=alt.X(
                "date:T",
                title=None,
                axis=alt.Axis(format="%Y"),
            ),
            y=alt.Y(
                f"{change_col}:Q",
                title="변화율 (%)",
            ),
            color=alt.Color(
                "방향:N",
                title=None,
                scale=alt.Scale(
                    domain=["증가", "감소"],
                    range=["#2E7D32", "#C62828"],
                ),
            ),
            tooltip=[
                alt.Tooltip(
                    "date:T",
                    title="기간",
                    format="%Y-%m",
                ),
                alt.Tooltip(
                    f"{change_col}:Q",
                    title=label,
                    format=".2f",
                ),
            ],
        )
        .properties(height=235)
    )

    return (bars + zero), label


def _market_block(name, data):
    data = _prepare(data)

    if data.empty:
        st.warning(f"{name} 데이터가 없습니다.")
        return

    summary = summarize_timeseries(data)
    compare = compare_recent_vs_longterm(data)

    frequency = summary.get(
        "frequency",
        "UNKNOWN",
    )

    latest = float(data.iloc[-1]["value"])

    c1, c2, c3 = st.columns(3)

    c1.metric(
        f"{name} 최신 시장규모",
        _million(latest),
    )

    c2.metric(
        "장기 CAGR",
        _pct(
            compare.get(
                "longterm_cagr_pct"
            )
        ),
    )

    c3.metric(
        "최근 5년 CAGR",
        _pct(
            compare.get(
                "recent_cagr_pct"
            )
        ),
    )

    st.altair_chart(
        _trend_chart(
            data,
            f"{name} 자동차 시장 추세",
        ),
        width="stretch",
    )

    change_chart, change_label = (
        _change_chart(data)
    )

    st.markdown(
        f"##### {change_label}"
    )

    st.altair_chart(
        change_chart,
        width="stretch",
    )

    growth = summary.get("max_growth")
    decline = summary.get("max_decline")

    if growth and decline:
        st.caption(
            f"최대 상승 {growth['period']} "
            f"{growth['yoy_pct']:+.2f}%  ·  "
            f"최대 하락 {decline['period']} "
            f"{decline['yoy_pct']:+.2f}%"
        )

    if frequency == "MONTHLY":
        st.caption(
            "월간 데이터 · MoM / YoY / "
            "3개월·12개월 이동평균 분석"
        )
    else:
        st.caption(
            "현재 원자료는 연간 데이터 · "
            "YoY / 3년 이동평균 분석"
        )

from pathlib import Path
import pandas as pd


BASE_DIR = Path(__file__).resolve().parents[1]
MARKET_MART = BASE_DIR / "data" / "mart" / "market_monthly.parquet"


def load_vehicle_timeseries(
    scope: str = "GLOBAL",
    metric: str = "VEHICLE_PRODUCTION_TOTAL",
) -> pd.DataFrame:
    """시장 mart에서 자동차 생산 시계열을 불러온다."""

    df = pd.read_parquet(MARKET_MART)

    ts = df[
        (df["market_scope"] == scope)
        & (df["metric"] == metric)
    ].copy()

    if ts.empty:
        return pd.DataFrame()

    ts["date"] = pd.to_datetime(
        ts["period"],
        format="%Y-%m",
        errors="coerce",
    )

    ts["value"] = pd.to_numeric(
        ts["value"],
        errors="coerce",
    )

    ts = (
        ts.dropna(subset=["date", "value"])
        .sort_values("date")
        .drop_duplicates(subset=["date"], keep="last")
        .reset_index(drop=True)
    )

    ts["year"] = ts["date"].dt.year
    ts["month"] = ts["date"].dt.month
    ts["period_label"] = ts["date"].dt.strftime("%Y-%m")

    return ts


def detect_frequency(df: pd.DataFrame) -> str:
    """
    실제 데이터의 시간 주기를 판별한다.

    같은 연도에 2개 이상의 월 데이터가 있으면 MONTHLY,
    연도별 1개 데이터만 있으면 ANNUAL로 판단한다.
    """

    if df.empty:
        return "UNKNOWN"

    temp = df.copy()

    if "date" not in temp.columns:
        return "UNKNOWN"

    counts = (
        temp.groupby(temp["date"].dt.year)
        ["date"]
        .count()
    )

    if not counts.empty and counts.max() >= 2:
        return "MONTHLY"

    return "ANNUAL"


def add_timeseries_features(
    df: pd.DataFrame,
) -> pd.DataFrame:
    """
    데이터 주기에 맞는 시계열 분석 변수를 추가한다.

    ANNUAL:
        YoY
        3년 이동평균
        3년 이동표준편차

    MONTHLY:
        MoM
        YoY
        3개월 이동평균
        12개월 이동평균
        12개월 이동표준편차
    """

    if df.empty:
        return df.copy()

    result = (
        df.sort_values("date")
        .reset_index(drop=True)
        .copy()
    )

    frequency = detect_frequency(result)
    result["frequency"] = frequency

    result["change_abs"] = result["value"].diff()

    if frequency == "MONTHLY":

        result["mom_pct"] = (
            result["value"]
            .pct_change(
                periods=1,
                fill_method=None,
            )
            * 100
        )

        result["yoy_pct"] = (
            result["value"]
            .pct_change(
                periods=12,
                fill_method=None,
            )
            * 100
        )

        result["ma_3m"] = (
            result["value"]
            .rolling(
                window=3,
                min_periods=1,
            )
            .mean()
        )

        result["ma_12m"] = (
            result["value"]
            .rolling(
                window=12,
                min_periods=1,
            )
            .mean()
        )

        result["std_12m"] = (
            result["value"]
            .rolling(
                window=12,
                min_periods=2,
            )
            .std()
        )

    else:

        result["mom_pct"] = pd.NA

        result["yoy_pct"] = (
            result["value"]
            .pct_change(
                periods=1,
                fill_method=None,
            )
            * 100
        )

        result["ma_3y"] = (
            result["value"]
            .rolling(
                window=3,
                min_periods=1,
            )
            .mean()
        )

        result["std_3y"] = (
            result["value"]
            .rolling(
                window=3,
                min_periods=2,
            )
            .std()
        )

    return result


def calculate_cagr(df: pd.DataFrame):
    """실제 날짜 간격을 사용하여 연환산 CAGR을 계산한다."""

    if df.empty or len(df) < 2:
        return None

    data = (
        df.sort_values("date")
        .dropna(subset=["date", "value"])
    )

    if len(data) < 2:
        return None

    first = float(data.iloc[0]["value"])
    last = float(data.iloc[-1]["value"])

    days = (
        data.iloc[-1]["date"]
        - data.iloc[0]["date"]
    ).days

    years = days / 365.2425

    if first <= 0 or last < 0 or years <= 0:
        return None

    return float(
        (
            (last / first) ** (1 / years)
            - 1
        )
        * 100
    )


def calculate_period_change(df: pd.DataFrame):
    """선택 기간 시작 대비 종료 변화율."""

    if df.empty or len(df) < 2:
        return None

    first = float(df.iloc[0]["value"])
    last = float(df.iloc[-1]["value"])

    if first == 0:
        return None

    return float(
        ((last / first) - 1) * 100
    )


def summarize_timeseries(df: pd.DataFrame) -> dict:
    """시계열 핵심 지표와 변곡점을 요약한다."""

    if df.empty:
        return {}

    analyzed = add_timeseries_features(df)
    frequency = detect_frequency(analyzed)

    valid_yoy = analyzed.dropna(
        subset=["yoy_pct"]
    )

    max_growth = None
    max_decline = None

    if not valid_yoy.empty:

        growth_row = valid_yoy.loc[
            valid_yoy["yoy_pct"].idxmax()
        ]

        decline_row = valid_yoy.loc[
            valid_yoy["yoy_pct"].idxmin()
        ]

        max_growth = {
            "period": growth_row["period_label"],
            "yoy_pct": float(
                growth_row["yoy_pct"]
            ),
        }

        max_decline = {
            "period": decline_row["period_label"],
            "yoy_pct": float(
                decline_row["yoy_pct"]
            ),
        }

    result = {
        "frequency": frequency,
        "start_period": analyzed.iloc[0]["period_label"],
        "end_period": analyzed.iloc[-1]["period_label"],
        "start_value": float(
            analyzed.iloc[0]["value"]
        ),
        "end_value": float(
            analyzed.iloc[-1]["value"]
        ),
        "cagr_pct": calculate_cagr(analyzed),
        "period_change_pct": calculate_period_change(
            analyzed
        ),
        "mean_yoy_pct": (
            float(valid_yoy["yoy_pct"].mean())
            if not valid_yoy.empty
            else None
        ),
        "yoy_volatility": (
            float(valid_yoy["yoy_pct"].std())
            if len(valid_yoy) > 1
            else None
        ),
        "max_growth": max_growth,
        "max_decline": max_decline,
    }

    if frequency == "MONTHLY":

        valid_mom = analyzed.dropna(
            subset=["mom_pct"]
        )

        result["mean_mom_pct"] = (
            float(valid_mom["mom_pct"].mean())
            if not valid_mom.empty
            else None
        )

        result["mom_volatility"] = (
            float(valid_mom["mom_pct"].std())
            if len(valid_mom) > 1
            else None
        )

    return result


def compare_recent_vs_longterm(
    df: pd.DataFrame,
    recent_years: int = 5,
) -> dict:
    """전체기간과 최근기간의 연환산 성장률을 비교한다."""

    if df.empty:
        return {}

    full = (
        df.sort_values("date")
        .reset_index(drop=True)
        .copy()
    )

    latest_date = full["date"].max()

    recent_start_date = (
        latest_date
        - pd.DateOffset(years=recent_years)
    )

    recent = full[
        full["date"] >= recent_start_date
    ].copy()

    return {
        "longterm_cagr_pct": calculate_cagr(full),
        "recent_cagr_pct": calculate_cagr(recent),
        "recent_start_period": (
            recent.iloc[0]["period_label"]
            if not recent.empty
            else None
        ),
        "recent_end_period": (
            recent.iloc[-1]["period_label"]
            if not recent.empty
            else None
        ),
    }


def analyze_market(
    scope: str,
    metric: str = "VEHICLE_PRODUCTION_TOTAL",
) -> dict:

    ts = load_vehicle_timeseries(
        scope=scope,
        metric=metric,
    )

    return {
        "frequency": detect_frequency(ts),
        "data": add_timeseries_features(ts),
        "summary": summarize_timeseries(ts),
        "trend_compare": compare_recent_vs_longterm(ts),
    }

from pathlib import Path

import pandas as pd

from config.settings import PATH_CONFIG
from config.schemas import MARKET_MONTHLY_COLUMNS
from loaders.kosis_loader import load_latest_vehicle_production_processed
from loaders.master_loader import load_market_assumption_master

def build_market_mart() -> pd.DataFrame:
    production = load_latest_vehicle_production_processed()

    rows = []

    # KOSIS 자동차 생산량을 시장 기준 데이터로 변환
    for _, row in production.iterrows():
        year = row.get("year")
        qty = row.get("production_qty")

        if pd.isna(year):
            continue

        country_code = str(row.get("country_code", ""))
        category = str(row.get("vehicle_category", ""))

        if country_code == "0":
            scope = "GLOBAL"
        elif country_code == "1005":
            scope = "KOREA"
        else:
            continue

        rows.append(
            {
                "period": f"{int(year)}-12",
                "market_scope": scope,
                "metric": f"VEHICLE_PRODUCTION_{category}",
                "value": (
                    float(qty)
                    if not pd.isna(qty)
                    else pd.NA
                ),
                "unit": "EA",
                "source_id": row.get("source_id"),
                "source_key": "kosis",
                "estimate_flag": row.get(
                    "estimate_flag",
                    "ACTUAL",
                ),
                "confidence_grade": "",
                "model_version": "",
            }
        )

    # 제품/당사 공급 기준은 Master를 단일 원본으로 사용
    assumptions = load_market_assumption_master()

    metric_map = {
        "coils_per_system": "COILS_PER_SYSTEM",
        "company_iv_per_system": "COMPANY_IV_PER_SYSTEM",
        "company_ov_per_system": "COMPANY_OV_PER_SYSTEM",
        "company_hsv_per_system": "COMPANY_HSV_PER_SYSTEM",
    }

    model_rows = []

    for _, row in assumptions.iterrows():
        model_id = row.get("model_id")
        parameter_name = row.get("parameter_name")

        if pd.isna(model_id) or parameter_name not in metric_map:
            continue

        value = pd.to_numeric(
            row.get("parameter_value"),
            errors="coerce",
        )

        model_rows.append(
            {
                "period": "2026-09",
                "market_scope": str(model_id),
                "metric": metric_map[parameter_name],
                "value": value if not pd.isna(value) else pd.NA,
                "unit": row.get("unit"),
                "source_id": row.get("source_id"),
                "source_key": "market_assumption_master",
                "estimate_flag": row.get("estimate_flag"),
                "confidence_grade": row.get("confidence_grade"),
                "model_version": row.get("model_version"),
            }
        )

    rows.extend(model_rows)

    mart = pd.DataFrame(rows)

    for column in MARKET_MONTHLY_COLUMNS:
        if column not in mart.columns:
            mart[column] = pd.NA

    mart = mart[MARKET_MONTHLY_COLUMNS]

    return mart


def save_market_mart() -> dict:
    mart = build_market_mart()

    OUTPUT_PATH.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    mart.to_parquet(
        OUTPUT_PATH,
        index=False,
    )

    return {
        "output_file": str(OUTPUT_PATH),
        "row_count": len(mart),
        "actual_count": int(
            (mart["estimate_flag"] == "ACTUAL").sum()
        ),
        "estimated_count": int(
            (mart["estimate_flag"] == "ESTIMATED").sum()
        ),
    }


if __name__ == "__main__":
    result = save_market_mart()

    print("Market mart build completed")
    print("OUTPUT_FILE:", result["output_file"])
    print("ROW_COUNT:", result["row_count"])
    print("ACTUAL:", result["actual_count"])
    print("ESTIMATED:", result["estimated_count"])
